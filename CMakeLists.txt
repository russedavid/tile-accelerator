cmake_minimum_required(VERSION 3.20)
project(tile_accelerator LANGUAGES CXX)
add_executable(tile-sim cpp/simulator.cpp)
target_compile_features(tile-sim PRIVATE cxx_std_17)
target_compile_options(tile-sim PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
option(TILE_SANITIZERS "Enable address/undefined sanitizers" OFF)
if(TILE_SANITIZERS)
  target_compile_options(tile-sim PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(tile-sim PRIVATE -fsanitize=address,undefined)
endif()
